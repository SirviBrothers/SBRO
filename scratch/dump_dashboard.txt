149:     async function renderHomeDashboard() {
150:         const sales = await StorageManager.getSales();
151:         const credits = await StorageManager.getCredits();
152:         
153:         let todaySales = 0;
154:         let weeklySales = 0;
155:         let monthlySales = 0;
156:         let totalRevenue = 0;
157:         
158:         const now = new Date();
159:         const startOfDay = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
160:         const startOfWeek = startOfDay - (now.getDay() * 24 * 60 * 60 * 1000); // Rough start of week
161:         const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1).getTime();
162: 
163:         sales.forEach(sale => {
164:             const saleDate = new Date(sale.date).getTime();
165:             totalRevenue += (sale.total || 0);
166:             
167:             if (saleDate >= startOfDay) todaySales += sale.total;
168:             if (saleDate >= startOfWeek) weeklySales += sale.total;
169:             if (saleDate >= startOfMonth) monthlySales += sale.total;
170:         });
171: 
172:         const avgOrder = sales.length > 0 ? (totalRevenue / sales.length) : 0;
173:         
174:         let totalDue = 0;
175:         credits.forEach(credit => {
176:             if (credit.status !== 'Paid') {
177:                 const totalPaid = credit.payments ? credit.payments.reduce((sum, p) => sum + p.amount, 0) : 0;
178:                 totalDue += ((credit.total || 0) - totalPaid);
179:             }
180:         });
181: 
182:         const eToday = document.getElementById('dash-daily-sales');
183:         if(eToday) eToday.textContent = `₹ ${todaySales.toFixed(2)}`;
184:         
185:         const eWeek = document.getElementById('dash-weekly-sales');
186:         if(eWeek) eWeek.textContent = `₹ ${weeklySales.toFixed(2)}`;
187:         
188:         const eMonth = document.getElementById('dash-monthly-sales');
189:         if(eMonth) eMonth.textContent = `₹ ${monthlySales.toFixed(2)}`;
190:         
191:         const eAvg = document.getElementById('dash-avg-order');
192:         if(eAvg) eAvg.textContent = `₹ ${avgOrder.toFixed(2)}`;
193:         
194:         const eDue = document.getElementById('dash-total-due');
195:         if(eDue) eDue.textContent = `₹ ${totalDue.toFixed(2)}`;
196:     }
